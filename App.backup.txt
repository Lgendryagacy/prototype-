// Helper function to get route from URL
const getRouteFromPath = (pathname: string): Route => {
  const path = pathname.replace('/', '') || 'home';

  // Site is now live - no countdown restrictions

  // Handle admin sub-routes
  if (path.startsWith('admin/ad-analytics/')) {
    return 'admin/ad-analytics';
  }

  // Handle dynamic brand routes (e.g., /brand/brand-name-id)
  if (path.startsWith('brand/')) {
    return 'fog-service-brand';
  }

  // Handle dynamic service routes (e.g., /service/service-name-id)
  if (path.startsWith('service/')) {
    return 'fog-service-detail';
  }

  // Handle dynamic job routes (e.g., /job/job-title-id)
  if (path.startsWith('job/')) {
    return 'job-detail';
  }

  // Handle dynamic user profile routes (e.g., /user/userId)
  if (path.startsWith('user/')) {
    return 'public-profile';
  }

  // Handle dynamic event detail routes (e.g., /event/event-name-id)
  if (path.startsWith('event/')) {
    return 'event-detail';
  }

  // Handle dynamic event participants routes (e.g., /event-participants/eventId)
  if (path.startsWith('event-participants/')) {
    return 'event-participants';
  }

  // Handle dynamic gig routes (e.g., /gig-detail/web-development-gigId)
  if (path.startsWith('gig-detail/')) {
    return 'gig-detail';
  }

  // Handle dynamic job applications routes (e.g., /job-applications/job-title-id)
  if (path.startsWith('job-applications/')) {
    return 'job-applications';
  }

  // Handle dynamic apply job routes (e.g., /apply-job/job-title-id)

    return 'apply-job';
  }

  if (path.startsWith('physical-job-application/')) {

// Helper function to extract ID from URL slug (e.g., "cool-brand-abc123" -> "abc123")
const extractIdFromSlug = (slug: string): string => {
  // Assume ID is the last part after the last hyphen
  // Format: brand-name-with-hyphens-ID
  const parts = slug.split('-');
  return parts[parts.length - 1];
};

// Helper function to create URL slug from name and ID
const createSlug = (name: string, id: string): string => {
  const sanitizedName = name
    .toLowerCase()
    .replace(/[^a-z0-9\s-]/g, '') // Remove special characters
    .replace(/\s+/g, '-') // Replace spaces with hyphens
    .replace(/-+/g, '-') // Replace multiple hyphens with single
    .trim();
  return `${sanitizedName}-${id}`;
};

        setIsAdmin(false);
        setIsAdminLoading(false);
        setUserRank(null);

        if (rankLoadingTimer) clearTimeout(rankLoadingTimer);
        if (popupTimer) clearTimeout(popupTimer);
      }
      setIsAuthLoading(false);
    });

    // Cleanup subscription and timers on unmount
    return () => {
      unsubscribe();
      if (rankLoadingTimer) clearTimeout(rankLoadingTimer);
      if (popupTimer) clearTimeout(popupTimer);
    };
  }, [isLoggedIn, isAuthLoading, rankLoadingTimer, popupTimer]);

  const handleLogin = () => {
    setIsLoggedIn(true);

    // Redirect to intended route after login
    if (intendedRoute) {

      setIntendedRoute(null);

  // Remove the handleSignupStarted function as it's no longer needed

  const handleLogout = async () => {

      await signOutUser();

      setIsAdmin(false);

    switch (route) {

      case 'customer-service': return <CustomerServicePage navigate={navigate} />;

        <UnifiedMessagesPage

        }

        return <AdminDashboardPage navigate={navigate} />;
      }
      case 'admin/ad-analytics': {
        // Simple admin check for analytics
        if (!isLoggedIn || !isAdmin) {
          return <HomePage navigate={navigate} isLoggedIn={isLoggedIn} />;
        }

        if (isAdminLoading) {

            <div className="min-h-screen flex items-center justify-center">

      }
      case '404': return <NotFoundPage navigate={navigate} />;
      case 'countdown': return <CountdownPage />;
      default: return <NotFoundPage navigate={navigate} />;
    }
  };

  return (
    <div className="min-h-screen flex flex-col bg-[--color-bg-primary] text-[--color-text-primary] transition-colors duration-300">
      <Toaster
        position="top-right"

        <Header
          currentTheme={currentTheme}
          setCurrentTheme={handleThemeChange}

          onLogout={handleLogout}

        />
      )}
      <main className={`flex-1 ${showFooter ? 'pb-20 md:pb-0' : ''}`}>
        {renderContent()}
      </main>
      {showFooter && <Footer navigate={navigate} />}
      <FloatingActionButton
        navigate={navigate}
        currentPath={window.location.pathname}
        currentUser={currentUser}
      />


      {/* ✅ Updated Rank Congratulations Modal with real rank */}
      {userRank !== null && (
        <RankCongratulationsModal
          isOpen={showCongratulationsModal}
          onClose={handleCloseCongratulationsModal}
          userRank={userRank}
          userName={currentUser?.displayName || currentUser?.email?.split('@')[0] || 'FOGSLY User'}
          isNewUser={false} // Always false since we don't distinguish between new/returning users anymore
        />
      )}

      {/* App Download Popup */}
      {showAppDownloadPopup && (
        <AppDownloadPopup onClose={() => setShowAppDownloadPopup(false)} />

      {/* TODO: Floating dock under working - temporarily disabled */}

        <FloatingDockNav onNavigate={(view) => navigate(view as Route)} />

    </div>
