import React, { useState, useEffect } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { 
  ArrowLeft, 
  ArrowRight, 
  Check, 
  User, 
  Briefcase, 
  FileText, 
  DollarSign,
  Star,
  Award,
  Clock,
  Target,
  X,
  Upload,
  Image as ImageIcon,
  Plus,
  Loader
} from 'lucide-react';
import { Button } from '../components/ui/Button';
import { Card, CardHeader, CardTitle, CardContent } from '../components/ui/Card';
import { Badge } from '../components/ui/Badge';
import { type Route } from '../types';
import { User as FirebaseUser } from '../firebase/types/user';
import { getUserProfile } from '../firebase/services/userService';
import { submitFreelanceRequest, getUserFreelanceRequest } from '../firebase/services/freelanceRequests';
import ImageCropper from '../components/ImageCropper';

interface BecomeFreelancerPageProps {
  navigate: (route: Route) => void;
  currentUser: FirebaseUser | null;
}

type Step = 1 | 2 | 3 | 4;

interface PortfolioItem {
  id: string;
  image: string; // URL or base64
  title: string;
  description: string;
  link: string;
}

interface FreelancerFormData {
  // Step 1: Personal Information
  fullName: string;
  title: string;
  bio: string;
  
  // Step 2: Skills & Expertise
  skills: string[];
  categories: string[];
  customCategory: string;
  experience: string;
  
  // Step 3: Portfolio & Work
  portfolioItems: PortfolioItem[];
  education: string;
  certifications: string[];
  
  // Step 4: Rates & Availability
  hourlyRate: number;
  currency: string;
  availability: string;
  languages: string[];
}

interface ValidationErrors {
  [key: string]: string;
}

const BecomeFreelancerPage: React.FC<BecomeFreelancerPageProps> = ({ navigate, currentUser }) => {
  const [currentStep, setCurrentStep] = useState<Step>(1);
  const [isLoading, setIsLoading] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [errors, setErrors] = useState<ValidationErrors>({});
  const [formData, setFormData] = useState<FreelancerFormData>({
    fullName: '',
    title: '',
    bio: '',
    skills: [],
    categories: [],
    customCategory: '',
    experience: 'beginner',
    portfolioItems: [],
    education: '',
    certifications: [],
    hourlyRate: 0,
    currency: 'PKR',
    availability: 'full-time',
    languages: ['English']
  });

  // Auto-fetch user profile data on mount
  useEffect(() => {
    if (currentUser) {
      fetchUserProfile();
      checkExistingRequest();
    }
  }, [currentUser]);

  const checkExistingRequest = async () => {
    if (!currentUser?.uid) return;
    
    try {
      const existingRequest = await getUserFreelanceRequest(currentUser.uid);
      if (existingRequest) {
        // User already has a request
        if (existingRequest.requestStatus === 'approved') {
          alert('Your freelancer profile has already been approved!');
          navigate('find-work');
        } else if (existingRequest.requestStatus === 'pending') {
          alert('Your freelancer application is pending review.');
          navigate('find-work');
        }
      }
    } catch (error) {
      console.error('Error checking existing request:', error);
    }
  };

  const fetchUserProfile = async () => {
    if (!currentUser?.uid) return;
    
    setIsLoading(true);
    try {
      const userProfile = await getUserProfile(currentUser.uid);
      
      if (userProfile) {
        setFormData(prev => ({
          ...prev,
          fullName: userProfile.name || currentUser.displayName || '',
          // Note: title is not in userProfile, keep empty
          skills: userProfile.skills || [],
        }));
      } else {
        // No profile exists, use basic info from auth
        setFormData(prev => ({
          ...prev,
          fullName: currentUser.displayName || '',
        }));
      }
    } catch (error) {
      console.error('Error fetching user profile:', error);
      // Use basic auth info as fallback
      setFormData(prev => ({
        ...prev,
        fullName: currentUser.displayName || '',
      }));
    } finally {
      setIsLoading(false);
    }
  };

  const steps = [
    { number: 1, title: 'Personal Info', icon: User },
    { number: 2, title: 'Skills & Expertise', icon: Star },
    { number: 3, title: 'Portfolio', icon: Briefcase },
    { number: 4, title: 'Rates & Availability', icon: DollarSign }
  ];

  const validateStep = (step: Step): boolean => {
    const newErrors: ValidationErrors = {};

    switch (step) {
      case 1:
        if (!formData.fullName.trim()) {
          newErrors.fullName = 'Full name is required';
        }
        if (!formData.title.trim()) {
          newErrors.title = 'Professional title is required';
        }
        if (!formData.bio.trim()) {
          newErrors.bio = 'Bio is required';
        } else if (formData.bio.length < 50) {
          newErrors.bio = 'Bio must be at least 50 characters';
        }
        break;

      case 2:
        if (formData.categories.length === 0) {
          newErrors.categories = 'Please select at least one category';
        }
        if (formData.skills.length < 3) {
          newErrors.skills = 'Please add at least 3 skills';
        }
        break;

      case 3:
        // Portfolio is optional, no validation needed
        break;

      case 4:
        if (!formData.hourlyRate || formData.hourlyRate <= 0) {
          newErrors.hourlyRate = 'Please enter a valid hourly rate';
        }
        if (formData.languages.length === 0) {
          newErrors.languages = 'Please select at least one language';
        }
        break;
    }

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleNext = () => {
    if (validateStep(currentStep)) {
      if (currentStep < 4) {
        setCurrentStep((currentStep + 1) as Step);
        setErrors({});
        window.scrollTo({ top: 0, behavior: 'smooth' });
      }
    } else {
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }
  };

  const handlePrevious = () => {
    if (currentStep > 1) {
      setCurrentStep((currentStep - 1) as Step);
      setErrors({});
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }
  };

  const handleSubmit = async () => {
    if (!validateStep(4)) {
      return;
    }
    
    if (!currentUser) {
      alert('You must be logged in to submit a freelancer request');
      navigate('auth');
      return;
    }
    
    setIsSubmitting(true);
    try {
      const requestData = {
        userId: currentUser.uid,
        userEmail: currentUser.email || '',
        userName: currentUser.displayName || formData.fullName,
        fullName: formData.fullName,
        title: formData.title,
        bio: formData.bio,
        skills: formData.skills,
        categories: formData.categories,
        experience: formData.experience,
        portfolioItems: formData.portfolioItems,
        education: formData.education,
        certifications: formData.certifications,
        hourlyRate: formData.hourlyRate,
        currency: formData.currency,
        availability: formData.availability,
        languages: formData.languages,
        requestStatus: 'pending' as const,
        
        // Initial stats (will be updated as freelancer works)
        totalReviews: 0,
        totalJobs: 0,
        totalEarnings: 0,
        averageRating: 0
      };

      const requestId = await submitFreelanceRequest(requestData);
      console.log('Freelancer request submitted successfully:', requestId);
      
      alert('Your freelancer application has been submitted! We will review it and get back to you soon.');
      navigate('freelancer-request-status');
    } catch (error) {
      console.error('Error submitting freelancer request:', error);
      alert('Failed to submit freelancer request. Please try again.');
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleSkillAdd = (skill: string) => {
    if (formData.skills.length >= 10) {
      setErrors(prev => ({ ...prev, skills: 'Maximum 10 skills allowed' }));
      return;
    }
    if (skill && !formData.skills.includes(skill)) {
      setFormData(prev => ({
        ...prev,
        skills: [...prev.skills, skill]
      }));
      if (formData.skills.length + 1 >= 3) {
        setErrors(prev => {
          const newErrors = { ...prev };
          delete newErrors.skills;
          return newErrors;
        });
      }
    }
  };

  const handleSkillRemove = (skillToRemove: string) => {
    setFormData(prev => ({
      ...prev,
      skills: prev.skills.filter(skill => skill !== skillToRemove)
    }));
  };

  const handleCategoryAdd = (category: string) => {
    if (formData.categories.length >= 5) {
      setErrors(prev => ({ ...prev, categories: 'Maximum 5 categories allowed' }));
      return;
    }
    if (category && !formData.categories.includes(category)) {
      setFormData(prev => ({
        ...prev,
        categories: [...prev.categories, category]
      }));
      setErrors(prev => {
        const newErrors = { ...prev };
        delete newErrors.categories;
        return newErrors;
      });
    }
  };

  const handleCategoryRemove = (categoryToRemove: string) => {
    setFormData(prev => ({
      ...prev,
      categories: prev.categories.filter(cat => cat !== categoryToRemove)
    }));
  };

  const handleAddPortfolioItem = () => {
    const newItem: PortfolioItem = {
      id: Date.now().toString(),
      image: '',
      title: '',
      description: '',
      link: ''
    };
    setFormData(prev => ({
      ...prev,
      portfolioItems: [...prev.portfolioItems, newItem]
    }));
  };

  const handleRemovePortfolioItem = (id: string) => {
    setFormData(prev => ({
      ...prev,
      portfolioItems: prev.portfolioItems.filter(item => item.id !== id)
    }));
  };

  const handlePortfolioItemChange = (id: string, field: keyof PortfolioItem, value: string) => {
    setFormData(prev => ({
      ...prev,
      portfolioItems: prev.portfolioItems.map(item =>
        item.id === id ? { ...item, [field]: value } : item
      )
    }));
  };

  const renderStepContent = () => {
    switch (currentStep) {
      case 1:
        return <Step1PersonalInfo formData={formData} setFormData={setFormData} errors={errors} />;
      case 2:
        return (
          <Step2SkillsExpertise 
            formData={formData} 
            setFormData={setFormData}
            onSkillAdd={handleSkillAdd}
            onSkillRemove={handleSkillRemove}
            onCategoryAdd={handleCategoryAdd}
            onCategoryRemove={handleCategoryRemove}
            errors={errors}
          />
        );
      case 3:
        return (
          <Step3Portfolio 
            formData={formData} 
            setFormData={setFormData}
            onAddItem={handleAddPortfolioItem}
            onRemoveItem={handleRemovePortfolioItem}
            onItemChange={handlePortfolioItemChange}
            errors={errors}
          />
        );
      case 4:
        return <Step4RatesAvailability formData={formData} setFormData={setFormData} errors={errors} />;
      default:
        return null;
    }
  };

  if (!currentUser) {
    return (
      <div className="min-h-screen bg-gradient-to-br from-[--color-bg-primary] to-[--color-bg-secondary] pt-20 px-4">
        <div className="max-w-2xl mx-auto text-center py-20">
          <Award className="w-20 h-20 mx-auto mb-6 text-[--color-primary]" />
          <h1 className="text-4xl font-bold mb-4">Join Fogsly as a Freelancer</h1>
          <p className="text-xl text-[--color-text-secondary] mb-8">
            Please log in to create your freelancer profile
          </p>
          <Button size="lg" onClick={() => navigate('auth')}>
            Log In to Continue
          </Button>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-gradient-to-br from-[--color-bg-primary] to-[--color-bg-secondary] pt-20 px-4 pb-20">
      <div className="max-w-4xl mx-auto">
        {/* Header */}
        <div className="mb-8">
          <button
            onClick={() => navigate('find-work')}
            className="flex items-center gap-2 text-[--color-text-secondary] hover:text-[--color-primary] mb-4 transition-colors"
          >
            <ArrowLeft className="w-4 h-4" />
            Back to Find Work
          </button>
          <h1 className="text-4xl font-bold mb-2">Become a Freelancer</h1>
          <p className="text-[--color-text-secondary]">
            Complete your profile to start getting hired on Fogsly
          </p>
        </div>

        {/* Progress Steps */}
        <div className="mb-8">
          <div className="flex items-center justify-between mb-4">
            {steps.map((step, index) => {
              const Icon = step.icon;
              const isCompleted = currentStep > step.number;
              const isCurrent = currentStep === step.number;

              return (
                <React.Fragment key={step.number}>
                  <div className="flex flex-col items-center flex-1">
                    <div
                      className={`w-12 h-12 rounded-full flex items-center justify-center mb-2 transition-all ${
                        isCompleted
                          ? 'bg-green-500 text-white'
                          : isCurrent
                          ? 'bg-[--color-primary] text-white'
                          : 'bg-[--color-bg-secondary] text-[--color-text-secondary] border-2 border-[--color-border]'
                      }`}
                    >
                      {isCompleted ? <Check className="w-6 h-6" /> : <Icon className="w-6 h-6" />}
                    </div>
                    <span
                      className={`text-sm font-medium text-center ${
                        isCurrent ? 'text-[--color-primary]' : 'text-[--color-text-secondary]'
                      }`}
                    >
                      {step.title}
                    </span>
                  </div>
                  {index < steps.length - 1 && (
                    <div
                      className={`h-1 flex-1 mx-2 transition-all ${
                        isCompleted ? 'bg-green-500' : 'bg-[--color-border]'
                      }`}
                    />
                  )}
                </React.Fragment>
              );
            })}
          </div>
        </div>

        {/* Step Content */}
        <AnimatePresence mode="wait">
          <motion.div
            key={currentStep}
            initial={{ opacity: 0, x: 20 }}
            animate={{ opacity: 1, x: 0 }}
            exit={{ opacity: 0, x: -20 }}
            transition={{ duration: 0.3 }}
          >
            {renderStepContent()}
          </motion.div>
        </AnimatePresence>

        {/* Navigation Buttons */}
        <div className="flex justify-between mt-8">
          <Button
            variant="outline"
            onClick={handlePrevious}
            disabled={currentStep === 1}
          >
            <ArrowLeft className="w-4 h-4 mr-2" />
            Previous
          </Button>
          
          {currentStep < 4 ? (
            <Button onClick={handleNext}>
              Next
              <ArrowRight className="w-4 h-4 ml-2" />
            </Button>
          ) : (
            <Button onClick={handleSubmit} disabled={isSubmitting}>
              {isSubmitting ? (
                <>
                  <Loader className="w-4 h-4 mr-2 animate-spin" />
                  Submitting...
                </>
              ) : (
                <>
                  <Check className="w-4 h-4 mr-2" />
                  Complete Profile
                </>
              )}
            </Button>
          )}
        </div>
      </div>
    </div>
  );
};

// Step 1: Personal Information

  formData: FreelancerFormData;
  setFormData: React.Dispatch<React.SetStateAction<FreelancerFormData>>;
  errors: ValidationErrors;
}> = ({ formData, setFormData, errors }) => {
  return (
    <Card>

        <CardTitle>Personal Information</CardTitle>

      <CardContent className="space-y-6">
        <div>
          <label className="block text-sm font-medium mb-2">

          </label>
          <input

          />

        </div>

        <div>
          <label className="block text-sm font-medium mb-2">
            Professional Title <span className="text-red-500">*</span>
          </label>
          <input
            type="text"
            value={formData.title}
            onChange={(e) => setFormData(prev => ({ ...prev, title: e.target.value }))}

            } bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary]`}

          </p>
        </div>

        <div>
          <label className="block text-sm font-medium mb-2">
            Professional Bio <span className="text-red-500">*</span>
          </label>
          <textarea
            value={formData.bio}
            onChange={(e) => setFormData(prev => ({ ...prev, bio: e.target.value }))}
            rows={6}
            maxLength={1000}
            className={`w-full px-4 py-2 rounded-lg border ${
              errors.bio ? 'border-red-500' : 'border-[--color-border]'
            } bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary]`}
            placeholder="Tell clients about your experience, expertise, and what makes you unique..."
          />
          {errors.bio && (
            <p className="text-red-500 text-sm mt-1">{errors.bio}</p>
          )}
          <p className="text-sm text-[--color-text-secondary] mt-1">
            {formData.bio.length}/1000 characters (minimum 50)
          </p>
        </div>
      </CardContent>
    </Card>
  );
};

// Step 2: Skills & Expertise
const Step2SkillsExpertise: React.FC<{
  formData: FreelancerFormData;
  setFormData: React.Dispatch<React.SetStateAction<FreelancerFormData>>;
  onSkillAdd: (skill: string) => void;
  onSkillRemove: (skill: string) => void;
  onCategoryAdd: (category: string) => void;
  onCategoryRemove: (category: string) => void;
  errors: ValidationErrors;
}> = ({ formData, setFormData, onSkillAdd, onSkillRemove, onCategoryAdd, onCategoryRemove, errors }) => {
  const [skillInput, setSkillInput] = useState('');
  const [selectedCategory, setSelectedCategory] = useState('');

  const predefinedCategories = [
    'Web Development',
    'Mobile Development',
    'Design',
    'Writing',
    'Marketing',
    'Data Science',
    'Video & Animation',
    'Music & Audio',
    'Business',
    'Lifestyle'
  ];

  const handleAddSkill = () => {
    if (skillInput.trim()) {
      onSkillAdd(skillInput.trim());
      setSkillInput('');
    }
  };

  const handleAddCategory = () => {
    if (selectedCategory === 'custom') {
      if (formData.customCategory.trim()) {
        onCategoryAdd(formData.customCategory.trim());
        setFormData(prev => ({ ...prev, customCategory: '' }));
        setSelectedCategory('');
      }
    } else if (selectedCategory) {
      onCategoryAdd(selectedCategory);
      setSelectedCategory('');
    }
  };

  return (
    <Card>
      <CardHeader>
        <CardTitle>Skills & Expertise</CardTitle>
      </CardHeader>
      <CardContent className="space-y-6">
        <div>
          <label className="block text-sm font-medium mb-2">
            Categories <span className="text-red-500">*</span> (Max 5)
          </label>
          <div className="flex gap-2 mb-3">
            <select
              value={selectedCategory}
              onChange={(e) => setSelectedCategory(e.target.value)}
              className="flex-1 px-4 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary]"
              disabled={formData.categories.length >= 5}
            >
              <option value="">Select a category...</option>
              {predefinedCategories.map(cat => (
                <option key={cat} value={cat} disabled={formData.categories.includes(cat)}>
                  {cat}
                </option>
              ))}
              <option value="custom">+ Add Custom Category</option>
            </select>
            <Button 
              onClick={handleAddCategory}
              disabled={!selectedCategory || formData.categories.length >= 5}
            >
              Add
            </Button>
          </div>

          {selectedCategory === 'custom' && (
            <input
              type="text"
              value={formData.customCategory}
              onChange={(e) => setFormData(prev => ({ ...prev, customCategory: e.target.value }))}
              className="w-full px-4 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary] mb-2"
              placeholder="Enter custom category name"
            />
          )}

          <div className="flex flex-wrap gap-2">
            {formData.categories.map(category => (
              <Badge
                key={category}
                className="px-3 py-1 flex items-center gap-2 bg-[--color-primary]/10 text-[--color-primary]"
              >
                {category}
                <button
                  onClick={() => onCategoryRemove(category)}
                  className="hover:text-red-500"
                >
                  <X className="w-3 h-3" />
                </button>
              </Badge>
            ))}
          </div>
          {errors.categories && (
            <p className="text-red-500 text-sm mt-2">{errors.categories}</p>
          )}
          <p className="text-sm text-[--color-text-secondary] mt-2">
            {formData.categories.length}/5 categories selected
          </p>
        </div>

        <div>
          <label className="block text-sm font-medium mb-2">
            Skills <span className="text-red-500">*</span> (Min 3, Max 10)
          </label>
          <div className="flex gap-2 mb-3">
            <input
              type="text"
              value={skillInput}
              onChange={(e) => setSkillInput(e.target.value)}
              onKeyPress={(e) => e.key === 'Enter' && (e.preventDefault(), handleAddSkill())}
              className="flex-1 px-4 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary]"
              placeholder="Add a skill (e.g., React, Photoshop, SEO)"
              disabled={formData.skills.length >= 10}
            />
            <Button onClick={handleAddSkill} disabled={formData.skills.length >= 10}>
              Add
            </Button>
          </div>
          <div className="flex flex-wrap gap-2">
            {formData.skills.map(skill => (
              <Badge
                key={skill}
                className="px-3 py-1 flex items-center gap-2 bg-blue-500/10 text-white"
              >
                {skill}
                <button
                  onClick={() => onSkillRemove(skill)}
                  className="hover:text-red-500"
                >
                  <X className="w-3 h-3" />
                </button>
              </Badge>
            ))}
          </div>
          {errors.skills && (
            <p className="text-red-500 text-sm mt-2">{errors.skills}</p>
          )}
          <p className="text-sm text-[--color-text-secondary] mt-2">
            {formData.skills.length}/10 skills (minimum 3 required)
          </p>
        </div>

        <div>
          <label className="block text-sm font-medium mb-2">
            Experience Level <span className="text-red-500">*</span>
          </label>
          <select
            value={formData.experience}
            onChange={(e) => setFormData(prev => ({ ...prev, experience: e.target.value }))}
            className="w-full px-4 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary]"
          >
            <option value="beginner">Beginner (0-2 years)</option>
            <option value="intermediate">Intermediate (2-5 years)</option>
            <option value="expert">Expert (5+ years)</option>
          </select>
        </div>
      </CardContent>
    </Card>
  );
};

// Step 3: Portfolio & Work
const Step3Portfolio: React.FC<{
  formData: FreelancerFormData;
  setFormData: React.Dispatch<React.SetStateAction<FreelancerFormData>>;
  onAddItem: () => void;
  onRemoveItem: (id: string) => void;
  onItemChange: (id: string, field: keyof PortfolioItem, value: string) => void;
  errors: ValidationErrors;
}> = ({ formData, setFormData, onAddItem, onRemoveItem, onItemChange, errors }) => {
  const [tempImageSrc, setTempImageSrc] = useState<string>('');
  const [cropItemId, setCropItemId] = useState<string>('');
  const [showCropper, setShowCropper] = useState(false);

  const handleImageSelect = (id: string, file: File) => {
    const reader = new FileReader();
    reader.onloadend = () => {
      setTempImageSrc(reader.result as string);
      setCropItemId(id);
      setShowCropper(true);
    };
    reader.readAsDataURL(file);
  };

  const handleCropComplete = (croppedImage: string) => {
    onItemChange(cropItemId, 'image', croppedImage);
    setShowCropper(false);
    setTempImageSrc('');
    setCropItemId('');
  };

  const handleCropCancel = () => {
    setShowCropper(false);
    setTempImageSrc('');
    setCropItemId('');
  };

  return (
    <>
      {showCropper && (
        <ImageCropper
          imageSrc={tempImageSrc}
          onCropComplete={handleCropComplete}
          onCancel={handleCropCancel}
          aspectRatio={16 / 9}
        />
      )}
      
      <Card>
        <CardHeader>
          <CardTitle>Portfolio & Education</CardTitle>
          <p className="text-sm text-[--color-text-secondary]">
            Showcase your best work (Optional) - Images will be displayed in 16:9 aspect ratio cards
          </p>
        </CardHeader>
        <CardContent className="space-y-6">
          <div>
            <div className="flex items-center justify-between mb-4">
              <label className="block text-sm font-medium">
                Portfolio Items
              </label>
              {formData.portfolioItems.length > 0 && (
                <Button onClick={onAddItem} variant="outline" size="sm">
                  <Plus className="w-4 h-4 mr-2" />
                  Add Item
                </Button>
              )}
            </div>
            
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              {formData.portfolioItems.length === 0 ? (
                <div className="col-span-full text-center py-16 border-2 border-dashed border-[--color-border] rounded-xl bg-gradient-to-br from-[--color-bg-secondary]/50 to-transparent">
                  <div className="max-w-sm mx-auto">
                    <div className="w-20 h-20 bg-[--color-primary]/10 rounded-full flex items-center justify-center mx-auto mb-4">
                      <ImageIcon className="w-10 h-10 text-[--color-primary]" />
                    </div>
                    <h4 className="font-semibold text-lg mb-2">No Portfolio Items Yet</h4>
                    <p className="text-[--color-text-secondary] mb-6 text-sm">
                      Add your best work to showcase your skills and attract clients. Each item includes an image, title, description, and optional link.
                    </p>
                    <Button onClick={onAddItem} className="gap-2">
                      <Plus className="w-4 h-4" />
                      Add Your First Portfolio Item
                    </Button>
                  </div>
                </div>
              ) : (
                <>
                  {formData.portfolioItems.map((item) => (
                    <div key={item.id} className="border-2 border-[--color-border] rounded-xl overflow-hidden bg-[--color-bg-primary] hover:border-[--color-primary]/50 transition-all">
                      {/* Portfolio Card Preview */}
                      <div className="relative group">
                        {item.image ? (
                          <>
                            <div className="aspect-video w-full bg-gradient-to-br from-gray-800 to-gray-900 overflow-hidden">
                              <img
                                src={item.image}
                                alt="Portfolio preview"
                                className="w-full h-full object-cover"
                              />
                            </div>
                            <button
                              onClick={() => onItemChange(item.id, 'image', '')}
                              className="absolute top-2 right-2 bg-red-500 text-white p-2 rounded-full hover:bg-red-600 opacity-0 group-hover:opacity-100 transition-opacity shadow-lg"
                              title="Remove image"
                            >
                              <X className="w-4 h-4" />
                            </button>
                            <button
                              onClick={() => {
                                // Trigger file input
                                const fileInput = document.getElementById(`file-${item.id}`) as HTMLInputElement;
                                fileInput?.click();
                              }}
                              className="absolute bottom-2 right-2 bg-white/90 text-gray-800 px-3 py-1.5 rounded-lg hover:bg-white text-sm font-medium opacity-0 group-hover:opacity-100 transition-opacity shadow-lg flex items-center gap-1"
                            >
                              <Upload className="w-3 h-3" />
                              Change
                            </button>
                          </>
                        ) : (
                          <label 
                            htmlFor={`file-${item.id}`}
                            className="aspect-video w-full flex flex-col items-center justify-center border-2 border-dashed border-[--color-border] bg-[--color-bg-secondary]/50 cursor-pointer hover:bg-[--color-bg-secondary] transition-colors"
                          >
                            <div className="w-12 h-12 bg-[--color-primary]/10 rounded-full flex items-center justify-center mb-3">
                              <Upload className="w-6 h-6 text-[--color-primary]" />
                            </div>
                            <span className="text-sm font-medium text-[--color-text-secondary]">
                              Upload Project Image
                            </span>
                            <span className="text-xs text-[--color-text-secondary] mt-1">
                              16:9 aspect ratio
                            </span>
                          </label>
                        )}
                        <input
                          id={`file-${item.id}`}
                          type="file"
                          accept="image/*"
                          onChange={(e) => {
                            const file = e.target.files?.[0];
                            if (file) handleImageSelect(item.id, file);
                          }}
                          className="hidden"
                        />
                      </div>

                      {/* Portfolio Item Details */}
                      <div className="p-4 space-y-3">
                        <div className="flex items-start justify-between gap-2">
                          <input
                            type="text"
                            value={item.title}
                            onChange={(e) => onItemChange(item.id, 'title', e.target.value)}
                            className="flex-1 px-3 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary] text-sm font-medium"
                            placeholder="Project Title"
                          />
                          <button
                            onClick={() => onRemoveItem(item.id)}
                            className="text-red-500 hover:text-red-600 p-2"
                            title="Remove portfolio item"
                          >
                            <X className="w-4 h-4" />
                          </button>
                        </div>

                        <textarea
                          value={item.description}
                          onChange={(e) => onItemChange(item.id, 'description', e.target.value)}
                          rows={2}
                          className="w-full px-3 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary] text-sm resize-none"
                          placeholder="Brief description of your project and role..."
                        />

                        <input
                          type="url"
                          value={item.link}
                          onChange={(e) => onItemChange(item.id, 'link', e.target.value)}
                          className="w-full px-3 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary] text-sm"
                          placeholder="🔗 Project link (optional)"
                        />
                      </div>
                    </div>
                  ))}
                </>
              )}
            </div>
          </div>

        <div>
          <label className="block text-sm font-medium mb-2">
            Education
          </label>
          <textarea
            value={formData.education}
            onChange={(e) => setFormData(prev => ({ ...prev, education: e.target.value }))}
            rows={4}
            className="w-full px-4 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary]"
            placeholder="e.g., Bachelor's in Computer Science - University Name (2018-2022)"
          />
        </div>

        <div>
          <label className="block text-sm font-medium mb-2">
            Certifications (Optional)
          </label>
          <textarea
            value={formData.certifications.join('\n')}
            onChange={(e) =>
              setFormData(prev => ({
                ...prev,
                certifications: e.target.value.split('\n').filter(c => c.trim())
              }))
            }
            rows={4}
            className="w-full px-4 py-2 rounded-lg border border-[--color-border] bg-[--color-bg-primary] focus:outline-none focus:ring-2 focus:ring-[--color-primary]"
            placeholder="List your certifications (one per line)&#10;e.g.,&#10;AWS Certified Solutions Architect&#10;Google Analytics Certification"
          />
        </div>
      </CardContent>
    </Card>
    </>
  );
};
