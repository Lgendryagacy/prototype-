import React, { useState, useEffect } from 'react';
import { useAuth } from '../firebase/hooks/useAuth';
import { getUserReferrals, createReferralCodeForUser } from '../firebase/services/referralService';
import { getUserData } from '../firebase/services/userService';
import { ReferralStats } from '../firebase/types/referral';
import { Users, TrendingUp, Calendar, Copy, CheckCircle, Gift, Share2, User, ChevronLeft, ChevronRight, ArrowLeft } from 'lucide-react';
import { motion } from 'framer-motion';


interface ReferralsPageProps {
    navigate: (route: any) => void;
    isLoggedIn: boolean;
    currentTheme: string;
    setCurrentTheme: (theme: any) => void;
    onLogout: () => void;
    currentUser: any;
}

const ReferralsPage: React.FC<Partial<ReferralsPageProps>> = ({
    navigate = () => { },
    isLoggedIn = true,
    currentTheme = 'light',
    setCurrentTheme = () => { },
    onLogout = () => { },
    currentUser = null
}) => {
    const { user } = useAuth();
    const [stats, setStats] = useState<ReferralStats | null>(null);
    const [referralCode, setReferralCode] = useState<string>('');
    const [loading, setLoading] = useState(true);
    const [copied, setCopied] = useState(false);
    const [generating, setGenerating] = useState(false);
    const [error, setError] = useState<string>('');

    // Pagination state
    const [currentPage, setCurrentPage] = useState(1);
    const itemsPerPage = 10;

    useEffect(() => {
        if (!user && !isLoggedIn) {
            navigate('auth');
            return;
        }

        if (user) {
            loadReferralData();
        }
    }, [user, isLoggedIn]);

    const loadReferralData = async () => {
        if (!user) {
            setLoading(false);
            return;
        }

        try {
            setLoading(true);
            setError('');

            // Check if user has referral code
            const userDoc = await getUserData(user.uid);

            if (!userDoc) {
                setError('User data not found. Please try again.');
                setLoading(false);
                return;
            }

            let code = userDoc?.referralCode || '';

            // Generate referral code if user doesn't have one
            if (!code) {
                setGenerating(true);
                try {
                    code = await createReferralCodeForUser(
                        user.uid,
                        userDoc?.name || '',
                        user.email || ''
                    );
                } catch (codeError) {
                    console.error('Error generating referral code:', codeError);
                    setError('Failed to generate referral code. Please try again later.');
                }
                setGenerating(false);
            }

            setReferralCode(code);

            // Load referral stats
            const referralStats = await getUserReferrals(user.uid);
            setStats(referralStats);
        } catch (error) {
            console.error('Error loading referral data:', error);
            setError('Failed to load referral data. Please refresh the page.');
        } finally {
            setLoading(false);
        }
    };

    const copyReferralCode = () => {
        navigator.clipboard.writeText(referralCode);
        setCopied(true);
        setTimeout(() => setCopied(false), 2000);
    };

    const shareReferralLink = () => {
        const referralLink = `${window.location.origin}/auth?ref=${referralCode}`;
        if (navigator.share) {
            navigator.share({
                title: 'Join FOGSLY with my referral code!',
                text: `Use my referral code ${referralCode} to join FOGSLY and unlock exclusive benefits!`,
                url: referralLink
            });
        } else {
            navigator.clipboard.writeText(referralLink);
            setCopied(true);
            setTimeout(() => setCopied(false), 2000);
        }
    };

    const formatDate = (date: Date) => {
        return new Date(date).toLocaleDateString('en-US', {
            year: 'numeric',
            month: 'short',
            day: 'numeric'
        });
    };

    // Pagination calculations
    const totalReferrals = stats?.referrals.length || 0;
    const totalPages = Math.ceil(totalReferrals / itemsPerPage);
    const startIndex = (currentPage - 1) * itemsPerPage;
    const endIndex = startIndex + itemsPerPage;
    const currentReferrals = stats?.referrals.slice(startIndex, endIndex) || [];

    const goToPage = (page: number) => {
        setCurrentPage(page);
    };

    const goToPreviousPage = () => {
        setCurrentPage(prev => Math.max(prev - 1, 1));
    };

    const goToNextPage = () => {
        setCurrentPage(prev => Math.min(prev + 1, totalPages));
    };

    // Reset to page 1 when stats change
    useEffect(() => {
        setCurrentPage(1);
    }, [stats?.referrals.length]);

    if (loading) {
        return (
            <div className="min-h-screen bg-gray-50 dark:bg-gray-900 pt-16">
                <div className="flex flex-col items-center justify-center py-20">
                    <div className="animate-spin rounded-full h-12 w-12 border-b-2 mb-4" style={{ borderColor: 'var(--color-primary)' }}></div>
                    <p className="text-gray-600 dark:text-gray-400">Loading your referral dashboard...</p>
                </div>
            </div>
        );
    }

    if (error) {
        return (
            <div className="min-h-screen bg-gray-50 dark:bg-gray-900 pt-16">
                <div className="max-w-2xl mx-auto px-4 py-20 text-center">
                    <div className="bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800 rounded-lg p-6">
                        <p className="text-red-600 dark:text-red-400 mb-4">{error}</p>
                        <button
                            onClick={() => loadReferralData()}
                            className="px-6 py-2 bg-[--color-primary] text-white rounded-lg hover:opacity-90 transition-opacity"
                        >
                            Try Again
                        </button>
                    </div>
                </div>
            </div>
        );
    }

            <div className="max-w-6xl mx-auto px-4 sm:px-6 lg:px-1 py-1 ">
                {/* Back Button */}
                <motion.div

                    transition={{ duration: 0.3 }}
                    className="mb-3"
                >
                    <button
                        onClick={() => navigate('home')}
                        className="inline-flex items-center gap-2 px-4 py-2 text-sm font-bold text-gray-900 dark:text-white hover:text-gray-600 dark:hover:text-gray-300 transition-colors"
                    >

        </div>

};

export default ReferralsPage;

// --- 3D Tilt Wrapper (Local Component) ---
const TiltCard = ({ children, className = "" }) => {
    const x = useMotionValue(0);
    const y = useMotionValue(0);
    const mouseX = useSpring(x, { stiffness: 400, damping: 30 });
    const mouseY = useSpring(y, { stiffness: 400, damping: 30 });

    function onMouseMove(event) {
        const { left, top, width, height } = event.currentTarget.getBoundingClientRect();
        x.set(event.clientX - left - width / 2);
        y.set(event.clientY - top - height / 2);
    }

    function onMouseLeave() {
        x.set(0);
        y.set(0);
    }

    const rotateX = useTransform(mouseY, [-300, 300], [2, -2]);
    const rotateY = useTransform(mouseX, [-300, 300], [-2, 2]);

    return (
        <motion.div
            onMouseMove={onMouseMove}
            onMouseLeave={onMouseLeave}
            style={{ rotateX, rotateY, transformStyle: "preserve-3d" }}
            className={`relative transition-all duration-300 ease-out ${className}`}
        >
            {children}
        </motion.div>
    );
};
