

import React, { useState, useEffect } from 'react';
import { motion } from 'framer-motion';
import { type Route } from '../types';
import { Card, CardContent, CardHeader, CardTitle } from '../components/ui/Card';
import { Button } from '../components/ui/Button';
import { Badge } from '../components/ui/Badge';
import { ArrowLeft, Star, Clock, ShoppingCart, MessageSquare, Check, ChevronDown, Eye } from 'lucide-react';
import { Accordion, AccordionItem, AccordionTrigger, AccordionContent } from '../components/ui/Accordion';
import { Textarea } from '../components/ui/Textarea';
import { getBrandServiceById } from '../firebase/services/brandServiceService';
import { getDoc, doc } from 'firebase/firestore';
import { db } from '../firebase/config';
import { BrandService } from '../firebase/types/service';
import { toast } from 'react-hot-toast';

const FogServiceDetailPage = ({ navigate, context: serviceId }: { navigate: (route: Route, context?: any, options?: { name?: string }) => void; context: any; }) => {
    const [service, setService] = useState<BrandService | null>(null);
    const [brandData, setBrandData] = useState<any>(null);
    const [loading, setLoading] = useState(true);
    const [activeImage, setActiveImage] = useState('');

    // Handle package selection
    const handlePackageSelect = (packageData: any) => {
        if (!service || !brandData) return;
        
        navigate('messages', {
            brandId: brandData.id,
            serviceInquiry: {
                serviceName: service.title,
                packageName: packageData.name,
                price: `PKR ${packageData.price.toLocaleString()}`,
                deliveryTime: packageData.deliveryTime
            }
        });
    };

    // Handle hourly rate selection
    const handleHourlyRateSelect = () => {
        if (!service || !brandData || service.pricing.type !== 'hourly' || !service.pricing.hourlyRate) return;
        
        navigate('messages', {
            brandId: brandData.id,
            serviceInquiry: {
                serviceName: service.title,
                hourlyRate: `PKR ${service.pricing.hourlyRate.rate.toLocaleString()}/hr`,
                minimumHours: service.pricing.hourlyRate.minimumHours
            }
        });
    };

    useEffect(() => {
        const fetchServiceData = async () => {
            try {
                setLoading(true);
                
                // Fetch service data
                const serviceData = await getBrandServiceById(serviceId);
                if (!serviceData) {
                    toast.error('Service not found');
                    navigate('fog-services');
                    return;
                }
                
                setService(serviceData);
                setActiveImage(serviceData.mainImage || serviceData.thumbnailImages?.[0] || '');
                
                // Fetch brand data
                const brandDoc = await getDoc(doc(db, 'brands', serviceData.brandId));
                if (brandDoc.exists()) {
                    setBrandData({ id: brandDoc.id, ...brandDoc.data() });
                }
            } catch (error) {
                console.error('Error fetching service:', error);
                toast.error('Failed to load service');
                navigate('fog-services');
            } finally {
                setLoading(false);
            }
        };

        if (serviceId) {
            fetchServiceData();
        }
    }, [serviceId]);

    if (loading) {
        return (
            <div className="bg-[--color-bg-primary] min-h-screen flex flex-col items-center justify-center">
                <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-[--color-primary] mb-4"></div>
                <p className="text-[--color-text-secondary]">Loading service...</p>
            </div>
        );
    }

    if (!service || !brandData) {
        return (
            <div className="bg-[--color-bg-primary] min-h-screen flex flex-col items-center justify-center">
                <h1 className="text-2xl font-bold mb-4">Service not found!</h1>
                <Button variant="ghost" onClick={() => navigate('fog-services')}>
                    <ArrowLeft className="w-4 h-4 mr-2" />
                    Back to All Services
                </Button>
            </div>
        );
    }
    
    const priceDisplay = service.pricing.type === 'contact'
        ? 'Contact for Pricing'
        : service.pricing.type === 'hourly'
        ? `PKR ${service.pricing.hourlyRate?.rate || 0}/hr`
        : service.pricing.packages && service.pricing.packages.length > 0
            ? `PKR ${Math.min(...service.pricing.packages.map(p => p.price)).toLocaleString()}`
            : 'PKR 0';
    
    const shortDescription = service.description.split('.').slice(0, 2).join('.') + '.';
    const deliveryTime = service.pricing.type === 'contact'
        ? 'Contact for details'
        : service.pricing.type === 'hourly' 
        ? `${service.pricing.hourlyRate?.minimumHours || 1} hours min`
        : service.pricing.packages?.[0]?.deliveryTime || 'Contact for details';

    return (
        <div className="bg-[--color-bg-primary]">
            <div className="container mx-auto px-4 py-8 pt-4">
                <div className="mb-2">
                    <Button variant="ghost" onClick={() => navigate('fog-services')}>
                        <ArrowLeft className="w-4 h-4 mr-2" />
                        Back to Services
                    </Button>
                </div>

                <div className="grid grid-cols-1 lg:grid-cols-12 gap-8 lg:gap-12">
                    {/* Main Content */}
                    <main className="lg:col-span-7 xl:col-span-8 space-y-4">
                        <motion.div initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }}>
                            <div className="flex flex-wrap gap-2 mb-1">
                                {service.categories.map((cat) => (
                                    <Badge key={cat} variant="secondary">{cat}</Badge>
                                ))}
                            </div>
                            <h1 className="text-3xl md:text-3xl font-bold">{service.title}</h1>
                            <div className="flex items-center flex-wrap gap-x-4 gap-y-2 mt-1 text-[--color-text-secondary]">
                                 <div className="flex items-center gap-1 text-yellow-500 font-bold">
                                    <Star size={16} className="fill-yellow-500" /> {service.stats.rating?.toFixed(1) || '5.0'}
                                    <span className="text-[--color-text-secondary] font-normal ml-1">({service.stats.reviews || 0} reviews)</span>
                                </div>
                                {/* <span className="text-[--color-text-secondary]">-</span> */}
                                {/* <div className="flex items-center gap-1">
                                    <Eye size={16} />
                                    <span>{service.stats.views || 0} views</span>
                                </div> */}
                                <span className="text-[--color-text-secondary]">-</span>
                                <span>
                                    By <Button variant="link" className="p-0 h-auto" onClick={() => navigate('fog-service-brand', brandData.id, { name: brandData.brandName })}>{brandData.brandName}</Button>
                                </span>
                            </div>
                        </motion.div>
                        
                         <motion.div initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4, delay: 0.1 }}>
                            <div >
                                {activeImage ? (
                                    <img src={activeImage} alt={service.title} className="rounded-lg w-full aspect-[4/2]  border border-[--color-border] object-cover"/>
                                ) : (
                                    <div className="rounded-lg w-full aspect-[2/1] bg-gradient-to-br from-[--color-primary]/20 to-[--color-primary]/10 flex items-center justify-center border border-[--color-border]">
                                        <span className="text-6xl font-bold text-[--color-primary] opacity-50">
                                            {service.title.split(' ').map(w => w[0]).join('').slice(0, 2)}
                                        </span>
                                    </div>
                                )}
                                {service.thumbnailImages && service.thumbnailImages.length > 0 && (
                                     <div className="grid grid-cols-5 gap-2 mt-2">
                                        {[service.mainImage, ...service.thumbnailImages].filter(Boolean).map((url, index) => (
                                            <button key={index} onClick={() => setActiveImage(url!)}>
                                                <img 
                                                    src={url!} 
                                                    alt={`Thumbnail ${index + 1}`} 
                                                    className={`rounded w-full aspect-[2/1] object-cover transition-all ${activeImage === url ? 'ring-2 ring-[--color-primary] ring-offset-2 ring-offset-[--color-bg-primary]' : 'opacity-60 hover:opacity-100'}`}
                                                />
                                            </button>
                                        ))}
                                    </div>
                                )}
                            </div>
                         </motion.div>
                         
                         <motion.div initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4, delay: 0.2 }}>
                            <Card>
                                <CardHeader><CardTitle>About This Service</CardTitle></CardHeader>
                                <CardContent>
                                    <p className="text-[--color-text-secondary] leading-relaxed whitespace-pre-wrap">{service.description}</p>
                                </CardContent>
                            </Card>
                        </motion.div>

                        {/* Packages Section - Show if service has packages */}
                        {service.pricing.type === 'fixed' && service.pricing.packages && service.pricing.packages.length > 0 && (
                            <motion.div initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4, delay: 0.22 }}>
                                <Card>
                                    <CardHeader>
                                        <CardTitle>
                                            {service.pricing.packages.length > 1 ? 'Compare Packages' : 'Package Details'}
                                        </CardTitle>
                                    </CardHeader>
                                    <CardContent>
                                        <div className={`grid gap-4 ${service.pricing.packages.length === 1 ? 'grid-cols-1' : service.pricing.packages.length === 2 ? 'grid-cols-1 md:grid-cols-2' : 'grid-cols-1 md:grid-cols-2 lg:grid-cols-3'}`}>
                                            {service.pricing.packages.map((pkg, index) => (
                                                <div 
                                                    key={pkg.id} 
                                                    className={`border border-[--color-border] rounded-lg p-6 hover:shadow-lg transition-all duration-300 flex flex-col ${
                                                        index === 1 && service.pricing.packages!.length === 3 
                                                            ? 'ring-2 ring-[--color-primary] relative' 
                                                            : ''
                                                    }`}
                                                >
                                                    {index === 1 && service.pricing.packages!.length === 3 && (
                                                        <div className="absolute -top-3 left-1/2 -translate-x-1/2">
                                                            <Badge className="bg-[--color-primary] text-white">Most Popular</Badge>
                                                        </div>
                                                    )}
                                                    
                                                    <div className="text-center mb-4">
                                                        <h3 className="text-xl font-bold mb-2">{pkg.name}</h3>
                                                        <div className="text-3xl font-bold text-[--color-primary] mb-1">
                                                            PKR {pkg.price.toLocaleString()}
                                                        </div>
                                                        <div className="flex items-center justify-center gap-2 text-sm text-[--color-text-secondary]">
                                                            <Clock size={14} />
                                                            <span>{pkg.deliveryTime}</span>
                                                        </div>
                                                    </div>

                                                    <div className="border-t border-[--color-border] pt-4 mb-4 flex-grow">
                                                        <p className="text-sm font-semibold mb-3 text-[--color-text-primary]">
                                                            What's Included:
                                                        </p>
                                                        <ul className="space-y-2">
                                                            {pkg.features.map((feature, idx) => (
                                                                <li 
                                                                    key={idx} 
                                                                    className="flex items-start gap-2 text-sm text-[--color-text-secondary]"
                                                                >
                                                                    <Check size={16} className="text-green-500 flex-shrink-0 mt-0.5" />
                                                                    <span>{feature}</span>
                                                                </li>
                                                            ))}
                                                        </ul>
                                                    </div>

                                                    <Button 
                                                        className="w-full mt-auto" 
                                                        variant={index === 1 && service.pricing.packages!.length === 3 ? 'default' : 'outline'}
                                                        onClick={() => handlePackageSelect(pkg)}
                                                    >
                                                        Select {pkg.name}
                                                    </Button>
                                                </div>
                                            ))}
                                        </div>
                                    </CardContent>
                                </Card>
                            </motion.div>
                        )}

                        <motion.div initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4, delay: 0.25 }}>
                             <Card>
                                <CardHeader><CardTitle>About The Brand</CardTitle></CardHeader>
                                <CardContent className="flex items-center gap-4">
                                    {brandData.logoUrl ? (
                                        <img src={brandData.logoUrl} alt={brandData.brandName} className="w-16 h-16 rounded-full object-cover flex-shrink-0" />
                                    ) : (
                                        <div className="w-16 h-16 rounded-full bg-[--color-bg-tertiary] flex-shrink-0 flex items-center justify-center text-2xl font-bold">
                                            {brandData.brandName.split(' ').map((n: string) => n[0]).join('')}
                                        </div>
                                    )}
                                    <div>
                                        <h3 className="font-bold text-lg">{brandData.brandName}</h3>
                                        <p className="text-sm text-[--color-text-secondary]">{brandData.tagline}</p>
                                        <Button variant="link" className="p-0 h-auto mt-1" onClick={() => navigate('fog-service-brand', brandData.id, { name: brandData.brandName })}>View Brand Page</Button>
                                    </div>
                                </CardContent>
                            </Card>
                        </motion.div>

                        {service.faqs && service.faqs.length > 0 && (
                            <motion.div initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4, delay: 0.3 }}>
                                <Card>
                                    <CardHeader><CardTitle>Frequently Asked Questions</CardTitle></CardHeader>
                                    <CardContent>
                                        <Accordion>
                                            {service.faqs.map((faq, index) => (
                                                <AccordionItem key={index} value={`item-${index}`}>
                                                    <AccordionTrigger>{faq.question}</AccordionTrigger>
                                                    <AccordionContent>{faq.answer}</AccordionContent>
                                                </AccordionItem>
                                            ))}
                                        </Accordion>
                                    </CardContent>
                                </Card>
                            </motion.div>
                        )}

                        {/* <CommentSection /> */}
                    </main>

                    {/* Sidebar */}
                    <aside className="lg:col-span-5 xl:col-span-4">
                         <motion.div initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4, delay: 0.35 }} className="sticky top-24">
                            <Card className="shadow-lg">
                                <CardHeader>
                                    <CardTitle>
                                        {service.pricing.type === 'contact' ? 'Custom Pricing' :
                                         service.pricing.type === 'hourly' ? 'Hourly Rate' : 
                                         service.pricing.packages?.[0]?.name || 'Standard Package'}
                                    </CardTitle>
                                </CardHeader>
                                <CardContent className="space-y-4">
                                    {service.pricing.type === 'contact' ? (
                                        <>
                                            <div className="text-center py-4">
                                                <div className="w-16 h-16 rounded-full bg-[--color-primary]/10 flex items-center justify-center mx-auto mb-3">
                                                    <MessageSquare className="w-8 h-8 text-[--color-primary]" />
                                                </div>
                                                <h3 className="text-2xl font-bold text-[--color-primary] mb-2">Contact for Pricing</h3>
                                                <p className="text-sm text-[--color-text-secondary]">
                                                    This service requires a custom quote based on your specific needs and requirements.
                                                </p>
                                            </div>
                                            <div className="border-t border-[--color-border] pt-4">
                                                <p className="text-sm text-[--color-text-secondary] pb-4">
                                                    {shortDescription}
                                                </p>
                                                <div className="bg-[--color-bg-secondary] rounded-lg p-4 mb-4">
                                                    <p className="text-sm font-semibold mb-2">Why custom pricing?</p>
                                                    <ul className="text-sm text-[--color-text-secondary] space-y-1">
                                                        <li>• Tailored to your project scope</li>
                                                        <li>• Flexible payment terms</li>
                                                        <li>• Professional consultation included</li>
                                                        <li>• Enterprise solutions available</li>
                                                    </ul>
                                                </div>
                                            </div>
                                            <div className="pt-2 space-y-3">
                                                <Button 
                                                    size="lg" 
                                                    className="w-full"
                                                    onClick={() => navigate('messages', { 
                                                        brandId: brandData.id,
                                                        serviceInquiry: {
                                                            serviceName: service.title,
                                                            contactForPricing: true
                                                        }
                                                    })}
                                                >
                                                    <MessageSquare size={16} className="mr-2"/>
                                                    Request Custom Quote
                                                </Button>
                                                <p className="text-xs text-center text-[--color-text-secondary]">
                                                    Get a personalized quote within 24 hours
                                                </p>
                                            </div>
                                        </>
                                    ) : (
                                        <>
                                            <h2 className="font-bold text-3xl text-[--color-primary]">{priceDisplay}</h2>
                                            <p className="text-sm text-[--color-text-secondary] pb-4 border-b border-[--color-border]">
                                                {shortDescription}
                                            </p>
                                            <div className="flex items-center gap-4 text-sm font-medium">
                                                <span className="flex items-center gap-2">
                                                    <Clock size={16} /> {deliveryTime}
                                                </span>
                                            </div>
                                            {service.pricing.type === 'fixed' && service.pricing.packages && service.pricing.packages[0] && (
                                                <div className="space-y-2">
                                                    <p className="text-sm font-semibold">What's Included:</p>
                                                    <ul className="text-sm space-y-2">
                                                        {service.pricing.packages[0].features.map((item, index) => (
                                                            <li key={index} className="flex items-center gap-2 text-[--color-text-secondary]">
                                                                <Check size={16} className="text-green-500"/> {item}
                                                            </li>
                                                        ))}
                                                    </ul>
                                                </div>
                                            )}
                                            {service.pricing.type === 'hourly' && service.pricing.hourlyRate && (
                                                <div className="space-y-2">
                                                    <p className="text-sm font-semibold">Details:</p>
                                                    <ul className="text-sm space-y-2 text-[--color-text-secondary]">
                                                        <li>• Minimum {service.pricing.hourlyRate.minimumHours} hours required</li>
                                                        <li>• Flexible scheduling available</li>
                                                    </ul>
                                                </div>
                                            )}
                                            <div className="pt-4 space-y-3">
                                                <Button 
                                                    size="lg" 
                                                    className="w-full"
                                                    onClick={() => {
                                                        if (service.pricing.type === 'hourly') {
                                                            handleHourlyRateSelect();
                                                        } else if (service.pricing.packages && service.pricing.packages[0]) {
                                                            handlePackageSelect(service.pricing.packages[0]);
                                                        }
                                                    }}
                                                >
                                                    Continue ({priceDisplay})
                                                </Button>
                                                <Button size="lg" variant="outline" className="w-full" onClick={() => navigate('messages', { brandId: brandData.id })}>
                                                    <MessageSquare size={16} className="mr-2"/> Message Brand
                                                </Button>
                                            </div>
                                        </>
                                    )}
                                </CardContent>
                            </Card>
                        </motion.div>
                    </aside>
                </div>
            </div>
        </div>
    );
};

const CommentSection = () => {
    const [comments, setComments] = useState([
      { id: 1, user: 'Sarah K.', avatar: 'SK', text: 'Does the final delivery include source files? I need them for my own designer.', time: '2 days ago' },
      { id: 2, user: 'Michael T.', avatar: 'MT', text: 'Incredible work! The logo they designed for my startup was exactly what I envisioned. Highly recommend.', time: '5 days ago' },
    ]);
    const [newComment, setNewComment] = useState('');

    const handleSubmit = (e: React.FormEvent) => {
      e.preventDefault();
      if (newComment.trim() === '') return;
      
      const commentToAdd = {
        id: Date.now(),
        user: 'Alex Johnson', // Mock logged-in user
        avatar: 'AJ',
        text: newComment,
        time: 'Just now',
      };
      setComments([commentToAdd, ...comments]);
      setNewComment('');
    };

    return (
      <motion.div initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4, delay: 0.35 }}>
        <Card>
          <CardHeader><CardTitle>Comments & Reviews ({comments.length})</CardTitle></CardHeader>
          <CardContent>
            <form onSubmit={handleSubmit} className="mb-6">
              <div className="flex items-start gap-4">
                <div className="w-10 h-10 rounded-full bg-[--color-bg-tertiary] flex items-center justify-center font-bold text-sm">AJ</div>
                <div className="flex-1">
                  <Textarea 
                    placeholder="Ask a question or leave a review..." 
                    value={newComment}
                    onChange={(e) => setNewComment(e.target.value)}
                  />
                  <Button type="submit" className="mt-2">Post Comment</Button>
                </div>
              </div>
            </form>
            <div className="space-y-6">
              {/* {comments.map(comment => (
                <div key={comment.id} className="flex items-start gap-4">
                  <div className="w-10 h-10 rounded-full bg-[--color-bg-tertiary] flex-shrink-0 flex items-center justify-center font-bold text-sm">{comment.avatar}</div>
                  <div>
                    <div className="flex items-center gap-2">
                      <p className="font-semibold">{comment.user}</p>
                      <p className="text-xs text-[--color-text-secondary]">{comment.time}</p>
                    </div>
                    <p className="text-[--color-text-secondary] mt-1">{comment.text}</p>
                  </div>
                </div>
              ))} */}
            </div>
          </CardContent>
        </Card>
      </motion.div>
    );
  };

export default FogServiceDetailPage;
