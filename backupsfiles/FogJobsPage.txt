const JobFeedSection = ({ navigate }: { navigate: (route: Route, context?: any) => void }) => {
    const [searchQuery, setSearchQuery] = useState('');
    const [filters, setFilters] = useState({
        country: '',
        city: '',
        category: 'All',
        experienceLevel: 'All',
        salaryRange: 'All',
    });
    const [sortBy, setSortBy] = useState('newest');

    const allCountries = useMemo(() => ['All', ...Array.from(new Set(onSiteJobs.map(j => j.country)))], []);
    const experienceLevels = ['All', 'Entry', 'Intermediate', 'Expert'];
    const salaryRanges = ['All', '$40k - $60k', '$60k - $100k', '$100k+'];
    const jobCategories = useMemo(() => ['All', ...Array.from(new Set(onSiteJobs.map(j => j.category)))], []);


    const handleFilterChange = (filterName: string, value: string) => {
        setFilters(prev => ({ ...prev, [filterName]: value }));
    };

    const filteredAndSortedJobs = useMemo(() => {
        let filtered = onSiteJobs.filter(job => {
            const searchLower = searchQuery.toLowerCase();
            const matchesSearch = searchLower === '' ||
                job.title.toLowerCase().includes(searchLower) ||
                job.companyName.toLowerCase().includes(searchLower) ||
                job.skills.some(s => s.toLowerCase().includes(searchLower));

            const matchesCategory = filters.category === 'All' || job.category === filters.category;
            const matchesExperience = filters.experienceLevel === 'All' || job.experienceLevel === filters.experienceLevel;
            const matchesCountry = filters.country === '' || job.country.toLowerCase().includes(filters.country.toLowerCase());
            const matchesCity = filters.city === '' || job.city.toLowerCase().includes(filters.city.toLowerCase());

            let matchesSalary = true;
            if (filters.salaryRange === '$40k - $60k') matchesSalary = job.salaryMin >= 40000 && job.salaryMax <= 60000;
            if (filters.salaryRange === '$60k - $100k') matchesSalary = job.salaryMin >= 60000 && job.salaryMax <= 100000;
            if (filters.salaryRange === '$100k+') matchesSalary = job.salaryMin >= 100000;

            return matchesSearch && matchesCategory && matchesExperience && matchesCountry && matchesCity && matchesSalary;
        });

        switch (sortBy) {
            case 'newest':
                filtered.sort((a, b) => b.postedDate.getTime() - a.postedDate.getTime());
                break;
            case 'highestPaying':
                filtered.sort((a, b) => b.salaryMin - a.salaryMin);
                break;
            case 'mostRelevant':
            default:
                break;
        }
        return filtered;
    }, [searchQuery, filters, sortBy]);

    return (
        <section className="py-24">
            <div className="container mx-auto px-4">
                <motion.div
                    initial={{ opacity: 0, y: 20 }}
                    whileInView={{ opacity: 1, y: 0 }}
                    viewport={{ once: true }}
                    transition={{ duration: 0.5 }}
                    className="text-center mb-12"
                >
                    <h2 className="text-3xl font-bold mb-4">Find Your Next On-Site Job</h2>
                </motion.div>

                <Card className="p-4 mb-8">
                    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4 items-center">
                        <div className="relative lg:col-span-4">
                            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-5 h-5 text-[--color-text-secondary]" />
                            <Input placeholder="Search by title, company, or skill..." value={searchQuery} onChange={e => setSearchQuery(e.target.value)} className="h-12 pl-10" />
                        </div>
                         <Input placeholder="Country" value={filters.country} onChange={e => handleFilterChange('country', e.target.value)} />
                         <Input placeholder="City" value={filters.city} onChange={e => handleFilterChange('city', e.target.value)} />
                        <Select value={filters.category} onValueChange={value => handleFilterChange('category', value)}>
                           <SelectTrigger><SelectValue placeholder="Category" /></SelectTrigger>
                            <SelectContent>
                                {jobCategories.map(cat => <SelectItem key={cat} value={cat}>{cat}</SelectItem>)}
                            </SelectContent>
                        </Select>
                         <Select value={filters.experienceLevel} onValueChange={value => handleFilterChange('experienceLevel', value)}>
                            <SelectTrigger><SelectValue placeholder="Experience" /></SelectTrigger>
                            <SelectContent>
                                {experienceLevels.map(lvl => <SelectItem key={lvl} value={lvl}>{lvl}</SelectItem>)}
                            </SelectContent>
                        </Select>
                        <Select value={filters.salaryRange} onValueChange={value => handleFilterChange('salaryRange', value)}>
                            <SelectTrigger><SelectValue placeholder="Salary Range" /></SelectTrigger>
                            <SelectContent>
                                {salaryRanges.map(range => <SelectItem key={range} value={range}>{range}</SelectItem>)}
                            </SelectContent>
                        </Select>
                    </div>
                </Card>

                <div className="flex justify-between items-center mb-6">
                    <h3 className="text-2xl font-bold">All Jobs ({filteredAndSortedJobs.length})</h3>
                    <div className="flex items-center gap-2">
                        <span className="text-sm text-[--color-text-secondary]">Sort by:</span>
                        <Select value={sortBy} onValueChange={setSortBy}>
                            <SelectTrigger className="w-[180px]"><SelectValue placeholder="Sort by" /></SelectTrigger>
                            <SelectContent>
                                <SelectItem value="newest">Newest</SelectItem>
                                <SelectItem value="highestPaying">Highest Paying</SelectItem>
                            </SelectContent>
                        </Select>
                    </div>
                </div>
                
                {filteredAndSortedJobs.length > 0 ? (
                    <div className="space-y-4">
                        {filteredAndSortedJobs.map((job) => (
                            <div key={job.id}><PhysicalJobCard job={job} navigate={navigate} /></div>
                        ))}
                    </div>
                ) : (
                    <div className="text-center py-16 bg-[--color-bg-secondary] rounded-lg">
                        <h3 className="text-xl font-semibold">No Jobs Found</h3>
                        <p className="text-[--color-text-secondary] mt-2">Try adjusting your search or filters to find what you're looking for.</p>
                    </div>
                )}
            </div>
        </section>
    )
};
