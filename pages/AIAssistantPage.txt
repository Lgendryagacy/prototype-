// ============== SERVICE INTENT DETECTION ==============
const SERVICE_KEYWORDS = [
  // e-service specific
  'e-service', 'e-services', 'fog service', 'fog services', 'fogsly service',
  'professional service', 'digital service', 'managed service', 'agency',
  // Buy / order intent
  'buy', 'order', 'purchase', 'package', 'plan', 'pricing', 'quote',
  'delivery', 'get done', 'done for me', 'done for you',
  // Service types
  'branding service', 'seo service', 'marketing service', 'design service',
  'logo design', 'video production', 'ad campaign', 'social media management',
  'content creation', 'brand identity', 'website development', 'app development',
  'product photography', 'copywriting service', 'email marketing',
];

function detectServiceIntent(message: string): { isMatch: boolean; keywords: string[] } {
  const lower = message.toLowerCase();
  const matched = SERVICE_KEYWORDS.filter(kw => lower.includes(kw));
  return { isMatch: matched.length > 0, keywords: matched };
}

function filterServices(services: BrandService[], keywords: string[]): BrandService[] {
  if (!keywords.length) return services.slice(0, 5);
  const scored = services.map(s => {
    const text = [
      s.title, s.description, s.brandName,
      ...(s.categories || []),
    ].join(' ').toLowerCase();
    const score = keywords.reduce((acc, kw) => acc + (text.includes(kw) ? 1 : 0), 0);
    return { s, score };
  });
  const matched = scored.filter(x => x.score > 0).sort((a, b) => b.score - a.score);
  const results = matched.length > 0 ? matched.map(x => x.s) : services;
  return results.slice(0, 5);
}

// ============== FREELANCER INTENT DETECTION ==============
const FREELANCER_KEYWORDS = [
  // Intent phrases (freelancer-specific)
  'hire', 'find freelancer', 'need freelancer', 'looking for freelancer', 'looking for developer',
  'looking for designer', 'looking for writer', 'who can', 'can you find',
  // Gig-related (strictly freelancer)
  'gig', 'gigs', 'freelancer', 'freelancers', 'talent', 'portfolio',
  // Roles
  'developer', 'designer', 'writer', 'marketer', 'editor', 'photographer',
  'programmer', 'coder', 'analyst', 'consultant', 'creator',
  'web developer', 'app developer', 'ui designer', 'ux designer', 'graphic designer',
  'full stack', 'backend', 'frontend', 'data analyst', 'copywriter',
  'illustrator', 'animator', 'video editor',
  // Tech/Skills
  'react', 'flutter', 'python', 'node', 'wordpress', 'javascript', 'typescript',
  'android', 'ios', 'mobile app',
  // Hire-related
  'hourly', 'hourly rate',
];

function detectFreelancerIntent(message: string): { isMatch: boolean; keywords: string[] } {
  const lower = message.toLowerCase();
  const matched = FREELANCER_KEYWORDS.filter(kw => lower.includes(kw));
  return { isMatch: matched.length > 0, keywords: matched };
}

function filterFreelancers(freelancers: ApprovedFreelancer[], keywords: string[]): ApprovedFreelancer[] {
  if (!keywords.length) return freelancers.slice(0, 4);
  const scored = freelancers.map(f => {
    const text = [
      f.title, f.bio,
      ...(f.skills || []),
      ...(f.categories || []),
    ].join(' ').toLowerCase();
    const score = keywords.reduce((acc, kw) => acc + (text.includes(kw) ? 1 : 0), 0);
    return { f, score };
  });
  const matched = scored.filter(s => s.score > 0).sort((a, b) => b.score - a.score);
  const results = matched.length > 0 ? matched.map(s => s.f) : freelancers;
  return results.slice(0, 5);
}

const AIAssistantPage: React.FC<AIAssistantPageProps> = ({ navigate, currentUser }) => {
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [inputMessage, setInputMessage] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const [showSuggestions, setShowSuggestions] = useState(true);
  const [suggestedFreelancers, setSuggestedFreelancers] = useState<ApprovedFreelancer[]>([]);
  const [isFetchingFreelancers, setIsFetchingFreelancers] = useState(false);
  const [allFreelancers, setAllFreelancers] = useState<ApprovedFreelancer[]>([]);
  const [suggestedServices, setSuggestedServices] = useState<BrandService[]>([]);
  const [allServices, setAllServices] = useState<BrandService[]>([]);
  const [sidebarTab, setSidebarTab] = useState<'freelancers' | 'services'>('freelancers');
  const messagesEndRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLTextAreaElement>(null);

  const suggestedQuestions = getSuggestedQuestions();

  // Pre-load all active freelancers & services once on mount
  useEffect(() => {
    getAllActiveFreelancers()
      .then(setAllFreelancers)
      .catch(console.error);

    // Fetch active Fog e-Services from Firestore
    const fetchServices = async () => {
      try {
        const q = query(
          collection(db, 'BrandsServices'),
          where('status', '==', 'active'),
          where('soldOut', '==', false),
          limit(100)
        );
        const snap = await getDocs(q);
        const fetched: BrandService[] = snap.docs.map(doc => ({
          id: doc.id,
          ...doc.data(),
          createdAt: doc.data().createdAt?.toDate?.() || new Date(),
          updatedAt: doc.data().updatedAt?.toDate?.() || new Date(),
        })) as BrandService[];
        setAllServices(fetched);
      } catch (err) {
        console.error('Error pre-loading services:', err);
      }
    };
    fetchServices();
  }, []);

  // Redirect if not logged in
  useEffect(() => {
    if (!currentUser) {
      navigate('auth');
    }
  }, [currentUser, navigate]);

  // Auto-scroll to bottom when messages change
  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  // Initialize with welcome message
  useEffect(() => {
    if (messages.length === 0) {
      setMessages([
        {
          role: 'assistant',
          content:
            "👋 Hello! I'm your Fogsly AI Assistant. I'm here to help you understand our platform and answer any questions about our modules, features, and earning opportunities. How can I assist you today?",
          timestamp: new Date(),
        },
      ]);
    }
  }, []);

  const handleSendMessage = async (messageText?: string) => {
    const textToSend = messageText || inputMessage.trim();
    if (!textToSend || isLoading) return;

    const userMessage: ChatMessage = {
      role: 'user',
      content: textToSend,
      timestamp: new Date(),
    };

    setMessages((prev) => [...prev, userMessage]);
    setInputMessage('');
    setIsLoading(true);
    setShowSuggestions(false);

    // Detect intent for freelancers and services in parallel
    const { isMatch: isFreelancerMatch, keywords: freelancerKws } = detectFreelancerIntent(textToSend);
    const { isMatch: isServiceMatch, keywords: serviceKws } = detectServiceIntent(textToSend);

    if (isFreelancerMatch) {
      setIsFetchingFreelancers(true);
      const results = filterFreelancers(allFreelancers, freelancerKws);
      setSuggestedFreelancers(results);
      setIsFetchingFreelancers(false);
    } else {
      setSuggestedFreelancers([]);
    }

    if (isServiceMatch) {
      const results = filterServices(allServices, serviceKws);
      setSuggestedServices(results);
    } else {
      setSuggestedServices([]);
    }

    // Auto-switch tab: whichever intent scored more keyword matches wins
    if (isServiceMatch && isFreelancerMatch) {
      setSidebarTab(serviceKws.length >= freelancerKws.length ? 'services' : 'freelancers');
    } else if (isServiceMatch) {
      setSidebarTab('services');
    } else if (isFreelancerMatch) {
      setSidebarTab('freelancers');
    }

    try {
      const response = await generateAIResponse(textToSend, messages);
      const assistantMessage: ChatMessage = {
        role: 'assistant',
        content: response,
        timestamp: new Date(),
      };
      setMessages((prev) => [...prev, assistantMessage]);
    } catch (error) {
      console.error('Error getting AI response:', error);
      const errorMessage: ChatMessage = {
        role: 'assistant',
        content:
          "I apologize, but I'm having trouble processing your request right now. Please try again in a moment.",
        timestamp: new Date(),
      };
      setMessages((prev) => [...prev, errorMessage]);
    } finally {
      setIsLoading(false);
    }
  };

  const handleSuggestionClick = (question: string) => {
    handleSendMessage(question);
  };

  const handleClearChat = () => {
    setMessages([
      {
        role: 'assistant',
        content:
          "Chat cleared! I'm ready to help you with any questions about Fogsly. What would you like to know?",
        timestamp: new Date(),
      },
    ]);
    setShowSuggestions(true);
  };

  const handleKeyPress = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      handleSendMessage();
    }
  };

  if (!currentUser) {
    return null; // Will redirect in useEffect
  }

  const getInitials = (name: string) =>
    name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2);

  return (
    <div className="fixed inset-0 flex flex-col bg-gradient-to-br from-[--color-bg-primary] via-[--color-bg-secondary] to-[--color-bg-primary]">
      <AnimatedBackground />

      {/* Header - Fixed at top */}
      <motion.div
        initial={{ opacity: 0, y: -20 }}
        animate={{ opacity: 1, y: 0 }}
        className="relative z-10 border-b border-white/10 backdrop-blur-xl bg-white/5 dark:bg-black/5"
      >
        <div className="container mx-auto px-4 py-4 max-w-6xl">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-3">
              <button
                onClick={() => navigate('home')}
                className="p-2 hover:bg-white/10 rounded-xl transition-colors"
                title="Back to Home"
              >
                <ArrowLeft className="w-5 h-5 text-[--color-text-secondary]" />
              </button>
              <div className="w-10 h-10 bg-gradient-to-br from-[--color-primary] to-[--color-accent] rounded-xl flex items-center justify-center shadow-lg">
                <Sparkles className="w-5 h-5 text-white" />
              </div>
              <div>
                <h1 className="text-xl md:text-2xl font-bold text-[--color-text-primary]">
                  Fogsly AI Assistant
                </h1>
                <p className="text-[--color-text-secondary] text-xs hidden sm:block">
                  Your intelligent guide to Fogsly
                </p>
              </div>
            </div>

            <button
              onClick={handleClearChat}
              className="flex items-center gap-2 px-3 py-2 bg-red-500/10 text-red-500 hover:bg-red-500/20 rounded-xl transition-colors text-sm"
              title="Clear chat"
            >
              <Trash2 className="w-4 h-4" />
              <span className="hidden sm:inline">Clear</span>
            </button>
          </div>
        </div>
      </motion.div>

      {/* Body: Chat + Sidebar */}
      <div className="relative z-10 flex flex-1 overflow-hidden">

        {/* Messages Area - Scrollable */}
        <div className="flex flex-col flex-1 overflow-hidden">
        <div className="flex-1 overflow-y-auto custom-scrollbar">
        <style>{`
          .custom-scrollbar::-webkit-scrollbar {
            width: 8px;
          }
          .custom-scrollbar::-webkit-scrollbar-track {
            background: transparent;
          }
          .custom-scrollbar::-webkit-scrollbar-thumb {
            background: rgba(var(--color-primary-rgb, 59, 130, 246), 0.3);
            border-radius: 10px;
          }
          .custom-scrollbar::-webkit-scrollbar-thumb:hover {
            background: rgba(var(--color-primary-rgb, 59, 130, 246), 0.5);
          }
          /* Firefox */
          .custom-scrollbar {
            scrollbar-width: thin;
            scrollbar-color: rgba(var(--color-primary-rgb, 59, 130, 246), 0.3) transparent;
          }
        `}</style>
        <div className="container mx-auto px-4 py-6 max-w-6xl">
          <div className="space-y-4 pb-4">
            <AnimatePresence>
              {messages.map((message, index) => (
                <motion.div
                  key={index}
                  initial={{ opacity: 0, y: 10 }}
                  animate={{ opacity: 1, y: 0 }}
                  exit={{ opacity: 0, y: -10 }}
                  className={`flex gap-3 ${
                    message.role === 'user' ? 'justify-end' : 'justify-start'
                  }`}
                >
                  {message.role === 'assistant' && (
                    <div className="w-8 h-8 sm:w-10 sm:h-10 bg-gradient-to-br from-[--color-primary] to-[--color-accent] rounded-full flex items-center justify-center flex-shrink-0">
                      <Bot className="w-4 h-4 sm:w-5 sm:h-5 text-white" />
                    </div>
                  )}

                  <div
                    className={`max-w-[85%] sm:max-w-[75%] md:max-w-[65%] rounded-2xl px-4 py-3 ${
                      message.role === 'user'
                        ? 'bg-gradient-to-br from-[--color-primary] to-[--color-accent] text-white shadow-lg'
                        : 'bg-white/20 dark:bg-black/20 text-[--color-text-primary] border border-white/10 backdrop-blur-sm'
                    }`}
                  >
                    <p className="whitespace-pre-wrap text-sm leading-relaxed">
                      {message.content}
                    </p>
                    <p
                      className={`text-xs mt-2 ${
                        message.role === 'user'
                          ? 'text-white/70'
                          : 'text-[--color-text-secondary]'
                      }`}
                    >
                      {message.timestamp.toLocaleTimeString([], {
                        hour: '2-digit',
                        minute: '2-digit',
                      })}
                    </p>
                  </div>

                  {message.role === 'user' && (
                    <div className="w-8 h-8 sm:w-10 sm:h-10 bg-gradient-to-br from-blue-500 to-purple-500 rounded-full flex items-center justify-center flex-shrink-0">
                      <User className="w-4 h-4 sm:w-5 sm:h-5 text-white" />
                    </div>
                  )}
                </motion.div>
              ))}
            </AnimatePresence>

            {/* Loading indicator */}
            {isLoading && (
              <motion.div
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                className="flex gap-3 justify-start"
              >
                <div className="w-8 h-8 sm:w-10 sm:h-10 bg-gradient-to-br from-[--color-primary] to-[--color-accent] rounded-full flex items-center justify-center">
                  <Bot className="w-4 h-4 sm:w-5 sm:h-5 text-white" />
                </div>
                <div className="bg-white/20 dark:bg-black/20 rounded-2xl px-4 py-3 border border-white/10 backdrop-blur-sm">
                  <div className="flex items-center gap-2">
                    <Loader2 className="w-4 h-4 animate-spin text-[--color-primary]" />
                    <span className="text-sm text-[--color-text-secondary]">
                      Thinking...
                    </span>
                  </div>
                </div>
              </motion.div>

            <div ref={messagesEndRef} />
          </div>

          {/* Suggested Questions - Show when chat is empty or cleared */}
          {showSuggestions && messages.length <= 1 && (
            <motion.div
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              className="pb-6"

              <p className="text-xs text-[--color-text-secondary] mb-3 flex items-center gap-1">
                <Zap className="w-3 h-3" />
                Try asking:
              </p>

                  <button
                    key={idx}
                    onClick={() => handleSuggestionClick(question)}

                <textarea

                  value={inputMessage}

                          <p className="text-[11px] text-[--color-text-secondary] truncate">{freelancer.title}</p>
                          <div className="flex items-center gap-2 mt-1">
                            <div className="flex items-center gap-0.5">
                              <Star className="w-3 h-3 text-yellow-400 fill-yellow-400" />
                              <span className="text-[11px] font-semibold text-[--color-text-secondary]">
                                {freelancer.averageRating ? freelancer.averageRating.toFixed(1) : '5.0'}
                              </span>
                            </div>
                            <span className="text-[--color-text-secondary]/40 text-[10px]">•</span>
                            <span className="text-[11px] font-bold text-[--color-primary]">PKR {freelancer.hourlyRate}/hr</span>
                          </div>
                        </div>
                        <ExternalLink className="w-3.5 h-3.5 text-[--color-text-secondary]/40 group-hover:text-[--color-primary] transition-colors flex-shrink-0 mt-1" />
                      </div>
                      {freelancer.skills?.length > 0 && (
                        <div className="flex flex-wrap gap-1 mt-2">
                          {freelancer.skills.slice(0, 3).map(skill => (
                            <span key={skill} className="text-[9px] px-2 py-0.5 rounded-full bg-[--color-primary]/10 text-[--color-primary] font-semibold uppercase tracking-wide">
                              {skill}
                            </span>
                          ))}
                        </div>
                      )}
                    </motion.div>
                  ))
                ) : (
                  suggestedServices.map((service, idx) => (
                    <motion.div
                      key={service.id}
                      initial={{ opacity: 0, y: 12 }}
                      animate={{ opacity: 1, y: 0 }}
                      transition={{ delay: idx * 0.07 }}
                      onClick={() => navigate('fog-service-detail', service.id, { name: service.title })}
                      className="group cursor-pointer rounded-2xl border border-white/10 bg-white/10 dark:bg-black/10 hover:bg-white/20 dark:hover:bg-black/20 hover:border-[--color-primary]/40 transition-all overflow-hidden"
                    >
                      {/* Service thumbnail */}
                      {service.mainImage && (
                        <div className="w-full h-24 overflow-hidden">
                          <img
                            src={service.mainImage}
                            alt={service.title}
                            className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
                          />
                        </div>
                      )}
                      <div className="p-3">
                        <div className="flex items-start justify-between gap-2">
                          <div className="flex-1 min-w-0">
                            <p className="text-sm font-bold text-[--color-text-primary] line-clamp-2 group-hover:text-[--color-primary] transition-colors leading-tight">
                              {service.title}
                            </p>
                            <p className="text-[11px] text-[--color-text-secondary] truncate mt-0.5">
                              by {service.brandName}
                            </p>
                          </div>
                          <ExternalLink className="w-3.5 h-3.5 text-[--color-text-secondary]/40 group-hover:text-[--color-primary] transition-colors flex-shrink-0 mt-0.5" />
                        </div>
                        <div className="flex items-center justify-between mt-2">
                          <div className="flex items-center gap-0.5">
                            <Star className="w-3 h-3 text-yellow-400 fill-yellow-400" />
                            <span className="text-[11px] font-semibold text-[--color-text-secondary]">
                              {service.stats?.rating ? service.stats.rating.toFixed(1) : '5.0'}
                            </span>
                          </div>
                          {service.pricing?.packages?.[0]?.price && (
                            <span className="text-[11px] font-bold text-[--color-primary]">
                              PKR {service.pricing.packages[0].price.toLocaleString()}
                            </span>
                          )}
                        </div>
                        {service.categories?.length > 0 && (
                          <div className="flex flex-wrap gap-1 mt-2">
                            {service.categories.slice(0, 2).map(cat => (
                              <span key={cat} className="text-[9px] px-2 py-0.5 rounded-full bg-purple-500/10 text-purple-400 font-semibold uppercase tracking-wide">
                                {cat}
                              </span>
                            ))}
                          </div>
                        )}
                      </div>
                    </motion.div>
                  ))
                )}
              </div>

              {/* View All Button */}
              <div className="sticky bottom-0 px-3 py-3 border-t border-white/10 bg-white/10 dark:bg-black/10 backdrop-blur-xl">
                {sidebarTab === 'freelancers' && suggestedFreelancers.length > 0 && (
                  <button
                    onClick={() => navigate('find-work' as Route)}
                    className="w-full py-2.5 rounded-xl bg-gradient-to-r from-[--color-primary] to-[--color-accent] text-white text-xs font-bold tracking-wide hover:opacity-90 transition-opacity flex items-center justify-center gap-2"
                  >
                    <Users className="w-3.5 h-3.5" />
                    Browse All Freelancers
                  </button>
                )}
                {sidebarTab === 'services' && suggestedServices.length > 0 && (
                  <button
                    onClick={() => navigate('fog-services' as Route)}
                    className="w-full py-2.5 rounded-xl bg-gradient-to-r from-purple-500 to-[--color-accent] text-white text-xs font-bold tracking-wide hover:opacity-90 transition-opacity flex items-center justify-center gap-2"
                  >
                    <ShoppingBag className="w-3.5 h-3.5" />
                    Browse All Services
                  </button>
                )}
              </div>
            </motion.div>
          )}
        </AnimatePresence>

      </div>{/* end body */}
    </div>
  );
};

export default AIAssistantPage;
